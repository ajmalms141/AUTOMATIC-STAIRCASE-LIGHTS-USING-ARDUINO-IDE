const int ledpin[]={2,3,4,5,6,7};
const int lednum=6;
const int irtop=9;
const int irbotm=8; 
int delaytime=500;
void setup() {
  for(int i=0;i<lednum;i++){
    pinMode(ledpin[i],OUTPUT);
    pinMode(irtop,INPUT);
    pinMode(irbotm,INPUT);
  }

}

void loop() {
  int irtopvalue=digitalRead(irtop);
  int irbotmvalue=digitalRead(irbotm);
if(irbotmvalue==LOW){
  for(int i=0;i<lednum;i++){
    digitalWrite(ledpin[i],HIGH);
    delay(delaytime);
  }
}
else if(irtopvalue==LOW){
  for(int i=lednum-1;i>=0;i--){
    digitalWrite(ledpin[i],LOW);
    delay(delaytime);
  } 
}

}
